// ---- PARCHE TEMPORAL: gripper 2 va a NG ----
if (gripperIndex == 2)
{
    bool colocada = false;

    for (int intento = 1; intento <= 18 && !colocada; intento++)
    {
        // 1) elegir primer buffer libre
        string libre = null;
        for (int b = 1; b <= 18; b++)
        {
            if (!GetRecipeValue("RSet", "NGbuffer" + b))
            {
                libre = string.Format("{0:d3}", b);
                break;
            }
        }
        if (libre == null)
        {
            MiddleLayer.LogF.AddLog(LogType.NGProduct,
                "Rack NG lleno - no hay buffer libre", true);
            return FCResultType.IDLE;
        }
        SysPara.ngPos = libre;

        // 2) mover a posicion NG para inspeccion
        if (!Robot.WriteToEpson("017", SysPara.ngPos, "FF", 20000))
            return FCResultType.IDLE;

        // 3) camara: verificar si el slot ya tiene pieza
        double exposure = GetSettingValue("MSet", "NGCCDCheckExpTime");
        if (!H1_NGCCDCheck.TakePicture(exposure))
            return FCResultType.IDLE;
        H1_NGCCDCheck.RunTB();

        if (Convert.ToBoolean(H1_NGCCDCheck.TB.Outputs["IsHaveProduct"].Value))
        {
            // ocupado: marcar y probar el siguiente
            DataTable dtng = RecipeData.Tables["RSet"];
            dtng.Rows[0]["NGbuffer" + Convert.ToInt32(SysPara.ngPos)] = true;
            dtng.AcceptChanges();
            this.WriteRecipeData(string.Format("{0}\\{1}.xml",
                SysPara.RecipeDataDirectory, SysPara.RecipeName));
            continue;
        }

        // 4) libre: soltar la pieza
        if (Robot.WriteToEpson("016", SysPara.ngPos,
            string.Format("{0:d2}", gripperIndex), 20000))
        {
            MiddleLayer.LogF.AddLog(LogType.NGProduct,
                "Gripper2 deshabilitado - pieza a NG " + SysPara.ngPos, true);
            colocada = true;
        }
        else
        {
            return FCResultType.IDLE;
        }
    }

    RunTM.Restart();
    return FCResultType.NEXT;
}
// ---- FIN PARCHE ----
